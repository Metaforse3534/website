INSTALLATION & TYPESCRIPT USAGE npm install @orbit/core 
import { Orbit, Agent } from '@orbit/core';

const orbit = new Orbit({ apiKey: process.env.ORBIT_API_KEY });

const execution = await orbit.run({
  intent: 'Generate test harness for distributed Postgres connector',
  agents: [Agent.RESEARCH, Agent.CODE],
  stream: true
});

for await (const token of execution.stream) {
  process.stdout.write(token);
}
